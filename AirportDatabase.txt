1. إنشاء قاعدة البيانات والجداول الأساسية

أ. إنشاء الجداول

-- جدول المسافرين
CREATE TABLE Passengers (
    PassengerID NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    FullName VARCHAR2(100) NOT NULL,
    PassportNumber VARCHAR2(50) UNIQUE NOT NULL,
    ContactNumber VARCHAR2(15) NOT NULL
);

-- جدول الرحلات
CREATE TABLE Flights (
    FlightID NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    FlightNumber VARCHAR2(50) UNIQUE NOT NULL,
    DepartureCity VARCHAR2(50) NOT NULL,
    ArrivalCity VARCHAR2(50) NOT NULL,
    DepartureDateTime DATE NOT NULL,
    ArrivalDateTime DATE NOT NULL,
    AvailableSeats NUMBER NOT NULL
);

-- جدول الحجوزات
CREATE TABLE Bookings (
    BookingID NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    PassengerID NUMBER NOT NULL,
    FlightID NUMBER NOT NULL,
    BookingDate DATE NOT NULL,
    CONSTRAINT FK_Passenger FOREIGN KEY (PassengerID) REFERENCES Passengers(PassengerID) ON DELETE CASCADE,
    CONSTRAINT FK_Flight FOREIGN KEY (FlightID) REFERENCES Flights(FlightID) ON DELETE CASCADE
);

-- جدول الموظفين
CREATE TABLE Employees (
    EmployeeID NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    FullName VARCHAR2(100) NOT NULL,
    JobTitle VARCHAR2(50) NOT NULL,
    ContactNumber VARCHAR2(15) NOT NULL
);

-- جدول المطارات
CREATE TABLE Airports (
    AirportID NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    AirportName VARCHAR2(100) NOT NULL,
    Location VARCHAR2(100) NOT NULL
);

-- جدول سجلات الأنشطة
CREATE TABLE ActivityLogs (
    LogID NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    Description VARCHAR2(255) NOT NULL,
    LogDate TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

3. الإجراءات (Procedures)

أ. إجراء لإضافة حجز جديد

CREATE OR REPLACE PROCEDURE AddBooking(
    p_PassengerID IN NUMBER,
    p_FlightID IN NUMBER,
    p_BookingDate IN DATE
) AS
    v_SeatsAvailable NUMBER;
BEGIN
    SELECT AvailableSeats INTO v_SeatsAvailable FROM Flights WHERE FlightID = p_FlightID;

    IF v_SeatsAvailable > 0 THEN
        INSERT INTO Bookings (PassengerID, FlightID, BookingDate)
        VALUES (p_PassengerID, p_FlightID, p_BookingDate);

        UPDATE Flights
        SET AvailableSeats = AvailableSeats - 1
        WHERE FlightID = p_FlightID;

        INSERT INTO ActivityLogs (Description)
        VALUES ('Booking added for PassengerID ' || p_PassengerID || ' on FlightID ' || p_FlightID);
    ELSE
        RAISE_APPLICATION_ERROR(-20001, 'No seats available for this flight.');
    END IF;
END;
/



ج. إجراء لحذف حجز

CREATE OR REPLACE PROCEDURE DeleteBooking(
    p_BookingID IN NUMBER
) AS
    v_FlightID NUMBER;
BEGIN
    SELECT FlightID INTO v_FlightID FROM Bookings WHERE BookingID = p_BookingID;

    DELETE FROM Bookings WHERE BookingID = p_BookingID;

    UPDATE Flights
    SET AvailableSeats = AvailableSeats + 1
    WHERE FlightID = v_FlightID;

    INSERT INTO ActivityLogs (Description)
    VALUES ('Booking deleted: BookingID ' || p_BookingID);
END;
/
CREATE OR REPLACE PROCEDURE DeletePassenger(

) AS
BEGIN
    DELETE FROM Passengers WHERE PassengerID = p_PassengerID;

    INSERT INTO ActivityLogs (Description)
    VALUES ('Passenger deleted: PassengerID ' || p_PassengerID);
END;

CREATE OR REPLACE PROCEDURE SearchTable(
    p_TableName IN VARCHAR2,
    p_ColumnName IN VARCHAR2,
    p_SearchValue IN VARCHAR2,
    p_Cursor OUT SYS_REFCURSOR
) AS
    v_Query VARCHAR2(4000);
BEGIN
    -- إنشاء الاستعلام الديناميكي للبحث
    v_Query := 'SELECT * FROM ' || p_TableName || ' WHERE ' || p_ColumnName || ' LIKE ''%' || p_SearchValue || '%''';
    
    -- فتح المؤشر لإرجاع النتائج
    OPEN p_Cursor FOR v_Query;
END;
/

4. الدوال (Functions)

أ. حساب عدد الحجوزات لمسافر معين

CREATE OR REPLACE FUNCTION GetPassengerBookings(
    p_PassengerID IN NUMBER
) RETURN NUMBER AS
    v_BookingCount NUMBER;
BEGIN
    SELECT COUNT(*) INTO v_BookingCount FROM Bookings WHERE PassengerID = p_PassengerID;
    RETURN v_BookingCount;
END;
/

ب. حساب عدد الرحلات المتاحة بين مدينتين

CREATE OR REPLACE FUNCTION GetAvailableFlights(
    p_DepartureCity IN VARCHAR2,
    p_ArrivalCity IN VARCHAR2
) RETURN NUMBER AS
    v_AvailableFlights NUMBER;
BEGIN
    SELECT COUNT(*) INTO v_AvailableFlights 
    FROM Flights 
    WHERE DepartureCity = p_DepartureCity AND ArrivalCity = p_ArrivalCity AND AvailableSeats > 0;
    RETURN v_AvailableFlights;
END;
/

ج. حساب عدد الموظفين بناءً على الوظيفة

CREATE OR REPLACE FUNCTION GetEmployeeCountByJob(
    p_JobTitle IN VARCHAR2
) RETURN NUMBER AS
    v_EmployeeCount NUMBER;
BEGIN
    SELECT COUNT(*) INTO v_EmployeeCount 
    FROM Employees 
    WHERE JobTitle = p_JobTitle;
    RETURN v_EmployeeCount;
END;
/

5. التريجرات (Triggers)

أ. تسجيل نشاط عند تعديل تفاصيل رحلة

CREATE OR REPLACE TRIGGER AfterFlightUpdate
AFTER UPDATE ON Flights
FOR EACH ROW
BEGIN
    INSERT INTO ActivityLogs (Description)
    VALUES ('Flight updated: FlightID ' || :OLD.FlightID || ', From ' || :OLD.DepartureCity || ' to ' || :OLD.ArrivalCity || ' updated to ' || :NEW.DepartureCity || ' to ' || :NEW.ArrivalCity);
END;
/

ب. تسجيل نشاط عند حذف حجز

CREATE OR REPLACE TRIGGER AfterBookingDelete
AFTER DELETE ON Bookings
FOR EACH ROW
BEGIN
    INSERT INTO ActivityLogs (Description)
    VALUES ('Booking deleted: BookingID ' || :OLD.BookingID);
END;
/

ج. تسجيل نشاط عند تعديل وظيفة موظف

CREATE OR REPLACE TRIGGER AfterEmployeeUpdate
AFTER UPDATE ON Employees
FOR EACH ROW
BEGIN
    INSERT INTO ActivityLogs (Description)
    VALUES ('Employee updated: EmployeeID ' || :OLD.EmployeeID || ', New Job Title: ' || :NEW.JobTitle);
END;
/

6. الحزمة (Package)

إنشاء الحزمة

CREATE OR REPLACE PACKAGE AirportPackage AS
    PROCEDURE AddAirport(p_AirportName IN VARCHAR2, p_Location IN VARCHAR2);
    PROCEDURE DeleteAirport(p_AirportID IN NUMBER);
    PROCEDURE UpdateAirport(p_AirportID IN NUMBER, p_NewName IN VARCHAR2, p_NewLocation IN VARCHAR2);
    PROCEDURE GetAllAirports;
END AirportPackage;
/

إنشاء جسم الحزمة (Package Body)

CREATE OR REPLACE PACKAGE BODY AirportPackage AS
    PROCEDURE AddAirport(p_AirportName IN VARCHAR2, p_Location IN VARCHAR2) IS
    BEGIN
        INSERT INTO Airports (AirportName, Location) VALUES (p_AirportName, p_Location);
        INSERT INTO ActivityLogs (Description) VALUES ('Airport added: ' || p_AirportName);
    END;

    PROCEDURE DeleteAirport(p_AirportID IN NUMBER) IS
    BEGIN
        DELETE FROM Airports WHERE AirportID = p_AirportID;
        INSERT INTO ActivityLogs (Description) VALUES ('Airport deleted: AirportID ' || p_AirportID);
    END;

    PROCEDURE UpdateAirport(p_AirportID IN NUMBER, p_NewName IN VARCHAR2, p_NewLocation IN VARCHAR2) IS
    BEGIN
        UPDATE Airports SET AirportName = p_NewName, Location = p_NewLocation WHERE AirportID = p_AirportID;
        INSERT INTO ActivityLogs (Description) VALUES ('Airport updated: AirportID ' || p_AirportID);
    END;

    PROCEDURE GetAllAirports IS
    BEGIN
        FOR rec IN (SELECT * FROM Airports) LOOP
            DBMS_OUTPUT.PUT_LINE(rec.AirportID || ' - ' || rec.AirportName || ' - ' || rec.Location);
        END LOOP;
    END;
END AirportPackage;
/
